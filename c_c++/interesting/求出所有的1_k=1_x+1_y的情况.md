//输入正整数k,求出所有的1/k=1/x+1/y的情况,其中正整数x>=y;
//由数学知识可以求出y的范围
#include"stdio.h"
#include"stdlib.h"
void main()
{int k,y,n=0;
    scanf("%d",&k);
    for(y=k+1;y<=k*2;y++)
    if((k*y%(y-k))==0)
    {printf("\n1/%d=1/%d+1/%d",k,k*y/(y-k),y);
        n++;
    }
    printf("\n一共有\t%d 种.\n",n);
    system("pause");
}
