/*有n个人围成一圈，顺序排号。从第一个人开始报数（从1到3报数），凡报到3的人退出  
圈子，问最后留下的是原来第几号的那位。*/  

```c
#include"stdio.h"
#include"stdlib.h"
int main()
{int i,j,k,n,a[1000];
    i=k=0; j=1;
    scanf("%d",&n);
    for(i;i<n;i++)
    a[i]=i+1;
    i=0;
    while(n-k>1)//n是总人数
    { //printf("j_%d  k_%d  i_%d\n",j,k,i);
        if((j==1||j==2)&&a[i]!=0)
        {j++;//计算还没有退出的人
            i++;}
        if(i==n)//如果到了一圈
        i=0;//i重新循环
        if(j==3&&a[i]!=0)
        {a[i]=0;
            j=1;   //如果数到三,退出,同时计数器j清零
            k++; i++;  }//k表示的是退出的人
        //看有没有到一圈
        if(a[i]==0) i++;
        if(i==n)//如果到了一圈
        i=0;
    }
    i=0;
    while(a[i]==0&&i<n) i++;
    printf("%d",a[i]);
    system("pause");
    return 1;}
```
