//输入整数一个,输出其阶乘从小到大的素数的指数

//如5!=2^3 * 3^1 * 5^1

```c
#include"stdio.h"
#include"stdlib.h"
int main()
{int a[1000],k=0;//存素数
    int b[1000],j,w;//存指数,w为输入
    for(int i=0;i<1000;i++)
    {a[i]=b[i]=0; }//初始化
    for(int i=2;i<1000;i++)
    {for( j=2;j<i;j++)
        if(i%j==0)
        break;
        if(j>=i)
        a[k++]=i;
    }//这部分计算出所有的素数,并存入a数组中
    j=1;
    scanf("%d",&w);
    for(int i=2;i<=w;i++)
    {j=j*i;
        for(int n=0;a[n]!=0&&(j>=a[n]);n++)
        while(j%a[n]==0)
        {j=j/a[n];
            b[n]++;
        }
    }//将得到的结果存入b数组之中,然后输出
    for(int i=0;b[i]!=0;i++)
    printf("%-3d",b[i]);
```
