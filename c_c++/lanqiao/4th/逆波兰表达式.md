/*正常的表达式称为中缀表达式，运算符在中间，主要是给人阅读的，机器求解并不方便。  
    例如：3 + 5 * (2 + 6) - 1  
    而且，常常需要用括号来改变运算次序。  
    相反，如果使用逆波兰表达式（前缀表达式）表示，上面的算式则表示为：  
    \- + 3 * 5 + 2 6 1  
    不再需要括号，机器可以用递归的方法很方便地求解。  
    为了简便，我们假设：  
    1\. 只有 + \- * 三种运算符  
    2\. 每个运算数都是一个小于10的非负整数  
    下面的程序对一个逆波兰表示串进行求值。  
    其返回值为一个结构：其中第一元素表示求值结果，第二个元素表示它已解析的字符数。*/  

```c
#include"stdio.h"
#include"stdlib.h"
struct EV
{
	int result;  //计算结果 
	int n;       //消耗掉的字符数 
};
struct EV evaluate(char* x)
{
	struct EV ev = {0,0};
	struct EV v1;
	struct EV v2;
	if(*x==0) return ev;//如果结束,返回
	if(x[0]>='0' && x[0]<='9'){
		ev.result = x[0]-'0';
		ev.n = 1;
		return ev;//实际上就是取当前EV中的一个值,就是将char转换为int然后返回
	}//如果是字符(运算符,或者空格) 就继续往下调用
	v1 = evaluate(x+1);//可以是根->左->右的方式,v1是左子树,v2是右子树,运算符是根
	v2 =evaluate(x+v1.n+1); //填空位置
	if(x[0]=='+') ev.result = v1.result + v2.result;
	if(x[0]=='*') ev.result = v1.result * v2.result;
	if(x[0]=='-') ev.result = v1.result - v2.result;
	ev.n = 1+v1.n+v2.n;//个数相加
	return ev;
}
int main()
{char x[]="-+3*5+261";
	struct EV ev;
	ev=evaluate(x);
printf("%d",ev.result);
printf("\n");
system("pause");
	return 0;}
```
